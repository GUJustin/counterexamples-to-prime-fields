\begin{corollary}[A second-order refinement of the gap bound]
\label{gm:refined-gap-field}
Fix a rational rate $\rho\in(0,1)$, and put
\[
 H=H_2(\rho),\qquad
 \gamma_G=\frac{3}{2\ln2}-2=0.16404256\ldots.
\]
For every sufficiently small $\varepsilon>0$, choose $n$ to be a
nearest multiple of the denominator of $\rho$ to
\[
 \frac{H}{\varepsilon^2(\log_2(1/\varepsilon)+\gamma_G)}.
\]
For every prime $p>n$, the interval-domain code
$C=\RS(\F_p,\{0,\ldots,n-1\},\rho n)$ satisfies
\begin{equation}\label{gm:refined-gap}
 \log_2\ell_C(1-\rho-\varepsilon)
 \ge \frac{H^2}{2\varepsilon^2
       (\log_2(1/\varepsilon)+\gamma_G)}
       -O_\rho(1/\varepsilon).
\end{equation}
The radius is strictly below the characteristic-based Elias radius
if $\varepsilon\log_2p>H_2(\rho+\varepsilon)$.
In particular, for every sufficiently large prime $p$, writing
$b=\log_2p$, there is such a code at a radius strictly below Elias with
\begin{equation}\label{gm:refined-field}
 \log_2\ell_C(1-\rho-\varepsilon)
 \ge \frac{b^2}{2(\log_2b-\log_2H+\gamma_G)}-O_\rho(b),
 \qquad
 \varepsilon=\frac{H}{b-H_2'(\rho)}.
\end{equation}
These bounds retain the leading statements of
Corollary~\ref{gm:gap-field}; they refine the next term and do not
remove its logarithmic denominator.
\end{corollary}
\begin{proof}
Set $m=\lceil\varepsilon n\rceil$ and $t=\rho n+m$.
The uniform growing-moment Gram estimate from
Section~\ref{sec:gram-ellipsoid}, converted to base-two logarithms,
reads
\[
 \log_2D_G=\frac{m^2}{2}
       \bigl(\log_2(n/m)+\gamma_G\bigr)
       +O_\rho(m\log n+m^4/n^2).
\]
Here either of the two proved Gram denominators may be used.
At the chosen length, $\log n\ll m\ll n$, and rounding $m$ or $n$
changes the displayed leading expression by $O_\rho(1/\varepsilon)$.
Also
$\log_2\binom nt=Hn+O_\rho(m+\log n)$.
Consequently the list bound is at least
\[
 Hn-\frac{\varepsilon^2n^2}{2}
       (\log_2(1/\varepsilon)+\gamma_G)
       -O_\rho(1/\varepsilon).
\]
The chosen $n$ maximizes this quadratic up to rounding and proves
\eqref{gm:refined-gap}. Since $m/n\ge\varepsilon$, list-size
monotonicity gives the claimed radius. The entropy estimate used in
Corollary~\ref{gm:gap-field} gives the stated Elias condition.

For the field-size assertion let $h=H_2'(\rho)$ and take
$\varepsilon=H/(b-h)$. For sufficiently large $b$ this is positive
and $\rho+\varepsilon<1$. Strict concavity gives
\[
 H_2(\rho+\varepsilon)<H+h\varepsilon
                         =b\varepsilon,
\]
so the radius is strictly below Elias. The chosen $n$ is
$\Theta_\rho(b^2/\log b)<p$. Substitution in
\eqref{gm:refined-gap} gives
\[
 \frac{(b-h)^2}
 {2(\log_2(b-h)-\log_2H+\gamma_G)}-O_\rho(b).
\]
Replacing $b-h$ by $b$ changes this by $O_\rho(b/\log b)$,
which is absorbed in the error in~\eqref{gm:refined-field}.
\end{proof}
